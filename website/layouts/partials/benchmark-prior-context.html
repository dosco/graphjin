{{- /* Prior-cohort context rows for a benchmark board. Paid eval runs land one model at a time, so the moment a new suite generation publishes its first row, every other model would vanish from the comparison surface. This partial carries forward, for each model absent from the current ranked cohort, its most recent legitimately demoted cohort row ("previous public benchmark cohort (…)") from the latest prior generation. Runs demoted for any other reason — retraction, scoring-contract supersession, harness defects — never qualify. Input: dict with "data" (the benchmark data file) and "ranked" (the current comparison cohort). Returns dict with "generation" (the prior cohort's generation, "" when none) and "runs" (context rows, best score first). */ -}} {{- $data := .data -}} {{- $ranked := .ranked -}} {{- $rankedModels := slice -}} {{- range $ranked -}}{{- $rankedModels = $rankedModels | append (string .model) -}}{{- end -}} {{- $candidates := slice -}} {{- range default (slice) $data.runs -}} {{- if and (eq (.accepted | default false) true) (strings.HasPrefix (.unranked_reason | default "") "previous public benchmark cohort") -}} {{- $candidates = $candidates | append . -}} {{- end -}} {{- end -}} {{- $priorGeneration := "" -}} {{- if gt (len $candidates) 0 -}} {{- range sort $candidates "generation" "desc" -}} {{- if eq $priorGeneration "" -}}{{- $priorGeneration = string .generation -}}{{- end -}} {{- end -}} {{- end -}} {{- $context := slice -}} {{- $seenModels := slice -}} {{- if ne $priorGeneration "" -}} {{- range sort (where $candidates "generation" $priorGeneration) "generated_at" "desc" -}} {{- if and (not (in $rankedModels (string .model))) (not (in $seenModels (string .model))) -}} {{- $context = $context | append . -}} {{- $seenModels = $seenModels | append (string .model) -}} {{- end -}} {{- end -}} {{- end -}} {{- if gt (len $context) 0 -}}{{- $context = sort $context "recall" "desc" -}}{{- end -}} {{- return (dict "generation" $priorGeneration "runs" $context) -}}
